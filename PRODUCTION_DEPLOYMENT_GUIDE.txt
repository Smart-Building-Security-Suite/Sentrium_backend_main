4. SSL/TLS CONFIGURATION

Option A: Let's Encrypt (Free, Recommended)
-------------------------------------------
# Install certbot
sudo apt install certbot python3-certbot-nginx

# Stop nginx temporarily
sudo systemctl stop nginx

# Generate certificate
sudo certbot certonly --standalone -d yourdomain.com

# Certificates will be at:
# /etc/letsencrypt/live/yourdomain.com/fullchain.pem
# /etc/letsencrypt/live/yourdomain.com/privkey.pem

# Convert to PKCS12 for Spring Boot
sudo openssl pkcs12 -export \
  -in /etc/letsencrypt/live/yourdomain.com/fullchain.pem \
  -inkey /etc/letsencrypt/live/yourdomain.com/privkey.pem \
  -out /etc/sentrium/keystore.p12 \
  -name tomcat \
  -passout pass:YOUR_KEYSTORE_PASSWORD

# Set permissions
sudo chown sentrium:sentrium /etc/sentrium/keystore.p12
sudo chmod 600 /etc/sentrium/keystore.p12

# Auto-renewal
sudo certbot renew --dry-run

Option B: Self-Signed (Testing Only)
------------------------------------
keytool -genkeypair -alias tomcat \
  -keyalg RSA -keysize 2048 \
  -storetype PKCS12 \
  -keystore keystore.p12 \
  -validity 365

5. DATABASE SETUP
=================================================================================

Create Database User:
--------------------
sudo -u postgres psql

CREATE USER sentrium_user WITH PASSWORD 'STRONG_PASSWORD_HERE';
CREATE DATABASE sentrium_production;
GRANT ALL PRIVILEGES ON DATABASE sentrium_production TO sentrium_user;
\c sentrium_production
GRANT ALL ON SCHEMA public TO sentrium_user;
\q

Configure PostgreSQL:
--------------------
# Edit /etc/postgresql/15/main/postgresql.conf
max_connections = 100
shared_buffers = 256MB
effective_cache_size = 1GB
work_mem = 16MB

# Edit /etc/postgresql/15/main/pg_hba.conf
# Add line:
host    sentrium_production    sentrium_user    127.0.0.1/32    md5

# Restart PostgreSQL
sudo systemctl restart postgresql

Database Backup Setup:
---------------------
# Create backup script: /usr/local/bin/backup_sentrium.sh
#!/bin/bash
BACKUP_DIR="/var/backups/sentrium"
DATE=$(date +%Y%m%d_%H%M%S)
mkdir -p $BACKUP_DIR

pg_dump -U sentrium_user sentrium_production | \
  gzip > $BACKUP_DIR/sentrium_$DATE.sql.gz

# Keep only last 30 days
find $BACKUP_DIR -name "sentrium_*.sql.gz" -mtime +30 -delete

# Add to crontab (daily at 3 AM)
0 3 * * * /usr/local/bin/backup_sentrium.sh

Alerting Rules (Prometheus):
----------------------------
# Create alert.rules.yml
groups:
  - name: application_alerts
    interval: 30s
    rules:
      - alert: HighErrorRate
        expr: rate(http_server_requests_seconds_count{status=~"5.."}[5m]) > 0.05
        for: 5m
        annotations:
          summary: "High error rate detected"

      - alert: DatabaseDown
        expr: up{job="spring-actuator"} == 0
        for: 2m
        annotations:
          summary: "Database connection failure"

      - alert: HighResponseTime
        expr: http_server_requests_seconds{quantile="0.99"} > 2
        for: 5m
        annotations:
          summary: "High response time (p99 > 2s)"

      - alert: LowDiskSpace
        expr: node_filesystem_avail_bytes{mountpoint="/var/sentrium"} < 5e9
        for: 5m
        annotations:
          summary: "Low disk space on video storage"

8. DEPLOYMENT STEPS
=================================================================================

Step 1: Create Application User
-------------------------------
sudo useradd -r -s /bin/false sentrium
sudo usermod -a -G video sentrium  # For FFmpeg

Step 2: Clone and Build Application
-----------------------------------
cd /opt
sudo git clone YOUR_REPO_URL sentrium
cd sentrium
sudo chown -R sentrium:sentrium /opt/sentrium

# Build application
./mvnw clean package -DskipTests

# Verify JAR created
ls -lh target/backend-*.jar

Step 3: Create Systemd Service
------------------------------
sudo nano /etc/systemd/system/sentrium.service

[Unit]
Description=Sentrium Security Suite Backend
After=network.target postgresql.service

[Service]
Type=simple
User=sentrium
Group=sentrium
WorkingDirectory=/opt/sentrium
ExecStart=/usr/bin/java \
  -Xms512m -Xmx2g \
  -Dspring.profiles.active=production \
  -jar /opt/sentrium/target/backend-0.0.1-SNAPSHOT.jar

Restart=always
RestartSec=10
StandardOutput=journal
StandardError=journal

# Security hardening
NoNewPrivileges=true
PrivateTmp=true
ProtectSystem=strict
ProtectHome=true
ReadWritePaths=/var/sentrium /var/log/sentrium

[Install]
WantedBy=multi-user.target

Step 4: Start Application
-------------------------
sudo systemctl daemon-reload
sudo systemctl enable sentrium
sudo systemctl start sentrium

# Check status
sudo systemctl status sentrium

# View logs
sudo journalctl -u sentrium -f

Step 5: Configure Nginx Reverse Proxy
-------------------------------------
sudo nano /etc/nginx/sites-available/sentrium

server {
    listen 80;
    server_name yourdomain.com;
    return 301 https://$server_name$request_uri;
}

server {
    listen 443 ssl http2;
    server_name yourdomain.com;

    ssl_certificate /etc/letsencrypt/live/yourdomain.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/yourdomain.com/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    client_max_body_size 100M;  # For video uploads

    location / {
        proxy_pass http://localhost:8080;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_connect_timeout 300;
        proxy_send_timeout 300;
        proxy_read_timeout 300;
    }

    # WebSocket support
    location /ws {
        proxy_pass http://localhost:8080;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection "upgrade";
        proxy_set_header Host $host;
    }
}

# Enable site
sudo ln -s /etc/nginx/sites-available/sentrium /etc/nginx/sites-enabled/
sudo nginx -t
sudo systemctl restart nginx

Step 6: Configure Firewall
--------------------------
sudo ufw allow 22/tcp     # SSH
sudo ufw allow 80/tcp     # HTTP (redirect to HTTPS)
sudo ufw allow 443/tcp    # HTTPS
sudo ufw enable

Scaling Considerations:
----------------------
# Horizontal Scaling:
- Deploy multiple instances behind load balancer
- Use shared database
- Use shared video storage (S3, NFS)
- Use Redis for session management

# Vertical Scaling:
- Increase heap size in systemd service
- Increase database connection pool
- Add more disk for video storage

# Database Optimization:
- Add indexes for slow queries
- Partition large tables
- Configure read replicas

Troubleshooting:
---------------
# Application won't start
sudo journalctl -u sentrium -n 100 --no-pager

# High memory usage
ps aux | grep java
sudo kill -3 <pid>  # Thread dump
sudo journalctl -u sentrium | grep -i "OutOfMemory"

# Database connection issues
sudo -u postgres psql -c "SELECT * FROM pg_stat_activity;"

# Video recording failures
sudo -u sentrium ffmpeg -version
ls -la /var/sentrium/videos
df -h /var/sentrium

Emergency Procedures:
--------------------
# Rollback deployment
sudo systemctl stop sentrium
cd /opt/sentrium
git checkout <previous-version-tag>
./mvnw clean package -DskipTests
sudo systemctl start sentrium

# Database recovery from backup
sudo systemctl stop sentrium
sudo -u postgres psql -c "DROP DATABASE sentrium_production;"
sudo -u postgres psql -c "CREATE DATABASE sentrium_production;"
gunzip -c /var/backups/sentrium/sentrium_YYYYMMDD.sql.gz | \
  sudo -u postgres psql sentrium_production
sudo systemctl start sentrium
